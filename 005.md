public static BoardState processCascade(BoardState currentState, Random random) {
    final List<Match> matches = findMatches(currentState.board());

    if (matches.isEmpty()) return currentState;

    return cascadePipeline(currentState, matches, random);
}

private static BoardState cascadePipeline(BoardState state, List<Match> matches, Random random) {
    return Pipe.of(state)
        .pipe(s -> removeMatches(s, matches))
        .pipe(s -> fillEmptySpaces(s, random))
        .pipe(s -> processCascade(s, random))
        .get();
}
