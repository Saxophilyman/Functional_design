Использование обёртки, позволяющий сымитировать конвейерный стиль:
```java
import java.util.function.Function;

public record Pipe<T>(T value) {

    public static <T> Pipe<T> of(T value) {
        return new Pipe<>(value);
    }

    public <R> Pipe<R> pipe(Function<T, R> function) {
        return new Pipe<>(function.apply(value));
    }

    public T get() {
        return value;
    }
}
```

public static BoardState processCascade(BoardState currentState, Random random) {
    final List<Match> matches = findMatches(currentState.board());

    if (matches.isEmpty()) return currentState;

    return Pipe.of(currentState)
        .pipe(state -> removeMatches(state, matches))
        .pipe(state -> fillEmptySpaces(state, random))
        .pipe(state -> processCascade(state, random))
        .get();
}
