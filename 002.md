public static BoardState processCascade(BoardState currentState, Random random) {
    List<Match> matches = findMatches(currentState.board());
    
    if (matches.isEmpty()) {
        return currentState;
    }

    BoardState afterRemoval = removeMatches(currentState, matches);
    BoardState afterFilling = fillEmptySpaces(afterRemoval, random);

    return processCascade(afterFilling, random);
}
